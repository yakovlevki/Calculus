\begin{theorem} [Теорема Пуассона] $\\$
    Пусть $f(x,y,z)=f(z)$ (зависит только от $z$) определена и непрерывна на сфере $x^2+y^2+z^2=R^2$. Тогда
    \[\iint\limits_{x^2+y^2+z^2=R^2}f(z)\ dS=2\pi R\cdot \int\limits_{-R}^{R}f(z)\ dz\]
\end{theorem}
\begin{proof}
    Параметризуем сферу с использованием цилиндрических координат
    \[\begin{cases}
        x=r\cdot\cos{\phi},\\
        y=r\cdot \sin{\phi},\\
        z=z
    \end{cases}\]
    При этом мы находимся на сфере
    \[x^2+y^2+z^2=R^2 \Rightarrow r^2+z^2=R^2 \Rightarrow r=\sqrt{R^2-z^2}\]
    Итак, искомая параметризация    
    \[\begin{cases}
        x=\sqrt{R^2-z^2}\cdot\cos{\phi},\\
        y=\sqrt{R^2-z^2}\cdot \sin{\phi},\\
        z=z 
    \end{cases}\]
    где $z\in[-R,R],\ \phi\in [0,2\pi]$.
    считаем частные производные:
    \[\left(\frac{-z}{\sqrt{R^2-z^2}}\cdot \cos{\phi}, \frac{-z}{\sqrt{R^2-z^2}}\sin{\phi}, 1\right),\ \ \left(-\sqrt{R^2-z^2}\cdot \sin{\phi}, \sqrt{R^2-z^2}\cdot \cos{\phi}, 0\right)\]
    Таким образом
    \[E=\frac{z^2}{R^2-z^2},\ G=R^2-z^2,\ F=0 \Rightarrow \sqrt{EG-F^2}=R\]
    тогда 
    \[\iint\limits_{x^2+y^2+z^2=R^2}f(z)\ dS=\int\limits_{0}^{2\pi}\int\limits_{-R}^{R}f(z)\cdot R\ dzd\phi=2\pi R\cdot \int\limits_{-R}^{R}f(z)\ dz\]
\end{proof}

\subsection{Элементы теории поля}
Пусть $\Omega\subset \R^3$ --- область, $f\in \mathcal{D}(\Omega)$. Напомним что оператор $\nabla f=(f'_x, f'_y, f'_z)$ называется градиентом.
\begin{definition}
    Пусть $\vec{u}=(u_1,u_2,u_3),\ \|\vec{u}\|=1$. Тогда производной функции $f$ по еаправлению $\vec{u}$ называется 
    \[\frac{f}{\vec{u}}(x_0,y_0,z_0)=\lim\limits_{t\to 0}\frac{f(x_0+tu_1, y_0+tu_2, z_0+tu_3)-f(x_0,y_0,z_0)}{t}=\nabla f\cdot \vec{u}\]
\end{definition}
% тут было бы неплохо пояснить почему градиент --- напрвление наибольшего роста функции
\begin{comm}
    Пусть $F(x,y,z)=0$ --- неявная функция:
    \begin{enumerate}
        \item $F\in \mathcal{C}^1(\Omega)$
        \item $F'_z(x_0,y_0,z_0)\neq 0$
    \end{enumerate} 
    тогда 
    \[z'_x=-\frac{F'_x}{F'_z},\ z'_y=-\frac{F'_y}{F'_z}\]
    подставим это в уравнение касательной
    \[z-z_0=z'_x(x_0,y_0)\cdot (x-x_0)+z'_y(x_0,y_0)(y-y_0)\]
    \[z-z_0=-\frac{F'_x}{F'_z}(x-x_0)-\frac{F'_y}{F'_z}(y-y_0)\]
    \[F'_x(x_0,y_0,z_0)(x-x_0)+F'_y(x_0,y_0,z_0)(y-y_0)+F'_z(x_0,y_0,z_0)(z-z_0)=0\]
    Итак, вектор градиента является вектором нормали к касательной плоскости.
\end{comm}

    Дивергенцией называется
